cmake_minimum_required(VERSION 3.25)
project(YYEngine VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(YY_BUILD_APP "Build SDL game applications" ON)
option(YY_BUILD_TESTS "Build deterministic tests" ON)

if(EMSCRIPTEN)
  # The one place the web toolchain version is pinned; CI installs this emsdk (docs/setup.md).
  file(STRINGS "${CMAKE_SOURCE_DIR}/cmake/web/emsdk-version.txt" YY_EMSDK_VERSION LIMIT_COUNT 1)
  if(NOT EMSCRIPTEN_VERSION VERSION_EQUAL YY_EMSDK_VERSION)
    message(FATAL_ERROR "Emscripten ${EMSCRIPTEN_VERSION} found, cmake/web/emsdk-version.txt pins ${YY_EMSDK_VERSION}")
  endif()
endif()

add_library(yy_core INTERFACE)
target_include_directories(yy_core INTERFACE engine/include)
add_library(tapdemo_logic games/tapdemo/src/model.cpp)
target_include_directories(tapdemo_logic PUBLIC games/tapdemo/include)
target_link_libraries(tapdemo_logic PUBLIC yy_core)

if(YY_BUILD_TESTS AND NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
  enable_testing()
  add_executable(yy_tests tests/core_tests.cpp tests/palette_tests.cpp tests/level_band_tests.cpp tests/level_generation_tests.cpp tests/font_tests.cpp games/tapdemo/src/game.cpp)
  target_link_libraries(yy_tests PRIVATE tapdemo_logic)
  target_compile_definitions(yy_tests PRIVATE YY_GAME_VERSION="test" YY_SOURCE_DIR="${CMAKE_SOURCE_DIR}")
  add_test(NAME engine_and_tapdemo COMMAND yy_tests)
  # Picks level seeds by simulated play (docs/levels.md); built, never run by the tests.
  add_executable(level_tune tests/level_tune.cpp)
  target_link_libraries(level_tune PRIVATE tapdemo_logic)
endif()

if(YY_BUILD_APP)
  include(FetchContent)
  set(SDL_SHARED OFF CACHE BOOL "" FORCE)
  set(SDL_STATIC ON CACHE BOOL "" FORCE)
  set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
  set(SDL_TESTS OFF CACHE BOOL "" FORCE)
  if(CMAKE_SYSTEM_NAME STREQUAL "iOS" OR EMSCRIPTEN)
    # Touch/audio games do not access cameras, Bluetooth controllers, or sensors.
    # Remove the SDK backends so Apple does not require unused privacy permissions
    # (and the browser build neither asks for them nor ships them).
    foreach(subsystem CAMERA JOYSTICK HAPTIC HIDAPI SENSOR)
      set(SDL_${subsystem} OFF CACHE BOOL "Unused on iOS" FORCE)
    endforeach()
  endif()
  # Fixed commit archive: no full Git history, and verify cached downloads too.
  set(YY_SDL_COMMIT 7f3ae3d57459e59943a4ecfefc8f6277ec6bf540)
  set(YY_SDL_SHA256 7a8347c770b90b33daac2352858ca03f9c9a2ccc8ce711054870361d1a6b32e5)
  set(SDL_REVISION "SDL-${YY_SDL_COMMIT}" CACHE STRING "Pinned SDL source revision" FORCE)
  set(YY_SDL_URL "https://codeload.github.com/libsdl-org/SDL/tar.gz/${YY_SDL_COMMIT}")
  if(DEFINED ENV{YY_SDL_ARCHIVE} AND NOT "$ENV{YY_SDL_ARCHIVE}" STREQUAL "")
    set(YY_SDL_URL "$ENV{YY_SDL_ARCHIVE}")
  endif()
  FetchContent_Declare(SDL3 URL "${YY_SDL_URL}" URL_HASH "SHA256=${YY_SDL_SHA256}"
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(SDL3)
  add_library(yy_runtime engine/src/runtime.cpp)
  target_link_libraries(yy_runtime PUBLIC yy_core SDL3::SDL3-static)
  if(WIN32)
    target_link_libraries(yy_runtime PRIVATE psapi)
  elseif(APPLE)
    target_link_libraries(yy_runtime PRIVATE "-framework Foundation")
    if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
      # SDL UIKit keyboard/mouse support needs this even with joysticks disabled.
      target_link_libraries(yy_runtime PRIVATE "-framework GameController")
    endif()
  endif()
  set(YY_GAME "" CACHE STRING "Select one game; empty builds all registered games")
  set(YY_BUNDLE_ID "" CACHE STRING "Selected game's app identifier override")
  set(YY_APP_VERSION "" CACHE STRING "Selected game's marketing version override")
  set(YY_BUILD_NUMBER "1" CACHE STRING "Unique distribution build number")
  include(cmake/games.cmake)
endif()
